import hashlib, json, os, subprocess, sys, tempfile, time
from pathlib import Path
repo=Path('/Users/bmilk/code/project/agent-loom')
evidence=repo/'agentloom-sdk/qa/evidence/round5-fixes-20261017'
root=Path(tempfile.mkdtemp(prefix='agentloom-round5-package-',dir='/private/tmp'))
commands=[]
def run(command, log):
    commands.append(command)
    result=subprocess.run(command,cwd=root,capture_output=True,text=True,timeout=90)
    (evidence/log).write_text(result.stdout+result.stderr)
    assert result.returncode==0, result.stderr
run([sys.executable,'-m','pip','wheel','--no-index','--no-deps','--no-build-isolation','--no-cache-dir',str(repo/'agentloom-sdk'),'-w',str(root/'wheels')],'wheel-build.txt')
wheel=next((root/'wheels').glob('*.whl'))
run([sys.executable,'-m','pip','install','--no-index','--no-deps','--no-compile','--target',str(root/'install'),str(wheel)],'wheel-install.txt')
modules=list((repo/'agentloom-sdk/src/agentloom').rglob('*.py'))
for module in modules:
    assert module.read_bytes()==(root/'install/agentloom'/module.relative_to(repo/'agentloom-sdk/src/agentloom')).read_bytes(),module
assert (root/'install/agentloom/py.typed').exists()
env={**os.environ,'PYTHONPATH':str(root/'install')}
check=subprocess.run([sys.executable,'-c','import agentloom; print(agentloom.__file__)'],cwd=root,env=env,capture_output=True,text=True,check=True)
assert check.stdout.strip()==str(root/'install/agentloom/__init__.py')
recoveries=[]
for kind in ['native-depth','native-depth-invalid']:
    work=root/kind
    work.mkdir()
    worker=repo/'agentloom-sdk/tests/support/completion_process.py'
    first=subprocess.run([sys.executable,str(worker),str(work),kind,'crash'],cwd=root,env=env,capture_output=True,text=True,timeout=20)
    assert first.returncode==74,first.stderr
    time.sleep(.4)
    second=subprocess.run([sys.executable,str(worker),str(work),kind,'resume'],cwd=root,env=env,capture_output=True,text=True,timeout=20)
    assert second.returncode==0,second.stderr
    outcome=json.loads(second.stdout)
    audit=(work/'audit').read_text().splitlines()
    assert outcome['status']=='succeeded' and audit.count('tool')==1 and audit.count('model')==2
    recoveries.append({'kind':kind,'first_exit_code':first.returncode,'result':outcome,'audit':audit})
summary={'commands':commands,'wheel':str(wheel),'sha256':hashlib.sha256(wheel.read_bytes()).hexdigest(),'source_modules':len(modules),'byte_equal_to_source_and_install':True,'import_path':check.stdout.strip(),'process_recovery':recoveries}
(evidence/'wheel-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps(summary,indent=2))
